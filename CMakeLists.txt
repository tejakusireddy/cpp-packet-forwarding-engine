cmake_minimum_required(VERSION 3.20)
project(packet_forwarding_engine VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(forwarding src/packet.cpp)
target_include_directories(forwarding PUBLIC include)
target_compile_features(forwarding PUBLIC cxx_std_20)
set_target_properties(forwarding PROPERTIES CXX_EXTENSIONS OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(forwarding PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(ENABLE_SANITIZERS)
    target_compile_options(forwarding PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(forwarding PUBLIC -fsanitize=address,undefined)
  endif()
elseif(ENABLE_SANITIZERS)
  message(FATAL_ERROR "ENABLE_SANITIZERS requires Clang or GCC")
endif()

if(BUILD_TESTING)
  add_executable(packet_tests tests/packet_tests.cpp)
  target_link_libraries(packet_tests PRIVATE forwarding)
  set_target_properties(packet_tests PROPERTIES CXX_EXTENSIONS OFF)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(packet_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  endif()
  add_test(NAME packet_tests COMMAND packet_tests)
endif()
