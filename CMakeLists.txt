cmake_minimum_required(VERSION 3.20)
project(packet_forwarding_engine VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
find_package(Threads REQUIRED)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENABLE_THREAD_SANITIZER "Enable the thread sanitizer" OFF)
option(BUILD_BENCHMARKS "Build reproducible forwarding benchmarks" ON)
if(ENABLE_SANITIZERS AND ENABLE_THREAD_SANITIZER)
  message(FATAL_ERROR "Address/undefined and thread sanitizers require separate builds")
endif()

if(BUILD_BENCHMARKS)
  add_executable(forwarding_bench benchmarks/forwarding_bench.cpp)
  target_link_libraries(forwarding_bench PRIVATE forwarding)
  set_target_properties(forwarding_bench PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(forwarding_bench PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()

add_library(forwarding src/packet.cpp src/fib.cpp src/next_hop.cpp src/forward.cpp src/flow.cpp src/snapshot.cpp src/ipc.cpp src/unix_socket.cpp src/routing_service.cpp)
target_include_directories(forwarding PUBLIC include)
target_compile_features(forwarding PUBLIC cxx_std_20)
target_link_libraries(forwarding PUBLIC Threads::Threads)
set_target_properties(forwarding PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(forwarding PRIVATE src/linux_io.cpp)
  add_executable(packet_forwarder src/main.cpp)
  target_link_libraries(packet_forwarder PRIVATE forwarding)
  set_target_properties(packet_forwarder PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(packet_forwarder PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(forwarding PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(ENABLE_SANITIZERS)
    target_compile_options(forwarding PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(forwarding PUBLIC -fsanitize=address,undefined)
  elseif(ENABLE_THREAD_SANITIZER)
    target_compile_options(forwarding PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(forwarding PUBLIC -fsanitize=thread)
  endif()
elseif(ENABLE_SANITIZERS OR ENABLE_THREAD_SANITIZER)
  message(FATAL_ERROR "Sanitizers require Clang or GCC")
endif()

if(BUILD_TESTING)
  foreach(test_name IN ITEMS packet_tests fib_tests next_hop_tests forward_tests flow_tests queue_tests worker_tests snapshot_tests control_tests ipc_tests unix_socket_tests routing_service_tests)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE forwarding)
    set_target_properties(${test_name} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(${test_name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  set_tests_properties(queue_tests worker_tests snapshot_tests control_tests ipc_tests unix_socket_tests routing_service_tests PROPERTIES TIMEOUT 60)
  if(BUILD_BENCHMARKS)
    foreach(mode IN ITEMS core queued)
      add_test(NAME benchmark_${mode}_smoke COMMAND forwarding_bench --mode ${mode}
               --workers 2 --packets 2048 --routes 128 --updates 256 --batch 32 --window 2)
      set_tests_properties(benchmark_${mode}_smoke PROPERTIES TIMEOUT 60)
    endforeach()
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(linux_io_tests tests/linux_io_tests.cpp)
    target_link_libraries(linux_io_tests PRIVATE forwarding)
    target_compile_options(linux_io_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    add_test(NAME linux_io_tests COMMAND linux_io_tests)
    set_tests_properties(linux_io_tests PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 10)
  endif()
endif()
